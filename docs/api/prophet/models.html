145    def fit(
146        self,
147        stan_init: dict[str, float], 
148        stan_data: dict[str, Any],
149        **kwargs: Any,
150    ) -> OrderedDict[str, np.ndarray]:
151        if 'inits' not in kwargs and 'init' in kwargs:
152            stan_init = self.sanitize_custom_inits(stan_init, kwargs['init'])
153            del kwargs['init']
154
155        inits_list, data_list = self.prepare_data(stan_init, stan_data)
156        args = dict(
157            data=data_list,
158            inits=inits_list,
159            algorithm='Newton' if cast(float, data_list['T']) < 100 else 'LBFGS',
160            iter=int(1e4),
161        )
162        args.update(kwargs)
163
164        try:
165            self.stan_fit = self.model.optimize(**args)
166        except RuntimeError as e:
167            # Fall back on Newton
168            if not self.newton_fallback or args['algorithm'] == 'Newton':
169                raise e
170            logger.warning('Optimization terminated abnormally. Falling back to Newton.')
171            args['algorithm'] = 'Newton'
172            self.stan_fit = self.model.optimize(**args)
173        params = self.stan_to_dict_numpy(
174            self.stan_fit.column_names, self.stan_fit.optimized_params_np)
175        for par in params:
176            params[par] = params[par].reshape((1, -1))
177        return params

179    def sampling(
180        self,
181        stan_init: dict[str, float],
182        stan_data: dict[str, Any],
183        samples: int,
184        **kwargs: Any,
185    ) -> OrderedDict[str, np.ndarray]:
186        if 'inits' not in kwargs and 'init' in kwargs:
187            stan_init = self.sanitize_custom_inits(stan_init, kwargs['init'])
188            del kwargs['init']
189
190        inits_list, data_list = self.prepare_data(stan_init, stan_data)
191        args = dict(
192            data=data_list,
193            inits=inits_list,
194        )
195        if 'chains' not in kwargs:
196            kwargs['chains'] = 4
197        iter_half = samples // 2
198        kwargs['iter_sampling'] = iter_half
199        if 'iter_warmup' not in kwargs:
200            kwargs['iter_warmup'] = iter_half
201        args.update(kwargs)
202
203        self.stan_fit = self.model.sample(**args)
204        res = self.stan_fit.draws()
205        (samples, c, columns) = res.shape
206        res = res.reshape((samples * c, columns))
207        params = self.stan_to_dict_numpy(self.stan_fit.column_names, res)
208
209        for par in params:
210            s = params[par].shape
211            if s[1] == 1:
212                params[par] = params[par].reshape((s[0],))
213
214            if par in ['delta', 'beta'] and len(s) < 2:
215                params[par] = params[par].reshape((-1, 1))
216
217        return params

235    def sanitize_custom_inits(
236        default_inits: dict[str, Any],
237        custom_inits: dict[str, Any],
238    ) -> dict[str, float | Any]:
239        """Validate that custom inits have the correct type and shape, otherwise use defaults."""
240        sanitized = {}

248                sanitized[param] = custom_inits[param]
249            else:
250                sanitized[param] = default_inits[param]
251        return sanitized

284    @staticmethod
285    def stan_to_dict_numpy(
286        column_names: Sequence[str],
287        data: np.ndarray,
288    ) -> OrderedDict[str, np.ndarray]:
289        output: OrderedDict[str, np.ndarray] = OrderedDict()
290
291        prev: str | None = None
292
293        start = 0
294        end = 0
295        two_dims = len(data.shape) > 1
296        for cname in column_names:
297            parsed = cname.split(".") if "." in cname else cname.split("[")
298            curr = parsed[0]
299            if prev is None:
300                prev = curr
301
302            if curr != prev:
303                if prev in output:
304                    raise RuntimeError(
305                        "Found repeated column name"
306                    )
307                if two_dims:
308                    output[prev] = np.array(data[:, start:end])
309                else:
310                    output[prev] = np.array(data[start:end])
311                prev = curr
312                start = end
313            end += 1
314        if prev in output:
315            raise RuntimeError(
316                "Found repeated column name"
317            )
318        assert prev is not None
319        if two_dims:
320            output[prev] = np.array(data[:, start:end])
321        else:
322            output[prev] = np.array(data[start:end])
323        return output
