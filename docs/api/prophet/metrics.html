 42def performance_metrics(
 43    df: pd.DataFrame,
 44    metrics: Sequence[str] | None = None,
 45    rolling_window: float = 0.1,
 46    monthly: bool = False,
 47) -> pd.DataFrame | None:
 48    """Compute performance metrics from cross-validation results.
 49
 50    Computes a suite of performance metrics on the output of cross-validation.
 51    By default, the following metrics are included:
 52    'mse': mean squared error
 53    'rmse': root mean squared error
 54    'mae': mean absolute error
 55    'mape': mean absolute percent error
 56    'mdape': median absolute percent error
 57    'smape': symmetric mean absolute percentage error
 58    'coverage': coverage of the upper and lower intervals
 59
 60    A subset of these can be specified by passing a list of names as the
 61    `metrics` argument.
 62
 63    Metrics are calculated over a rolling window of cross validation
 64    predictions, after sorting by horizon. Averaging is first done within each
 65    value of horizon, and then across horizons as needed to reach the window
 66    size. The size of that window (number of simulated forecast points) is
 67    determined by the rolling_window argument, which specifies a proportion of
 68    simulated forecast points to include in each window. rolling_window=0 will
 69    compute it separately for each horizon. The default of rolling_window=0.1
 70    will use 10% of the rows in df in each window. rolling_window=1 will
 71    compute the metric across all simulated forecast points. The results are
 72    set to the right edge of the window.
 73
 74    If rolling_window < 0, then metrics are computed at each datapoint with no
 75    averaging (i.e., 'mse' will actually be squared error with no mean).
 76
 77    The output is a dataframe containing column 'horizon' along with columns
 78    for each of the metrics computed.
 79
 80    Parameters
 81    ----------
 82    df: The dataframe returned by cross_validation.
 83    metrics: Performance metrics to compute. A sequence of names; if not
 84        provided, uses ['mse', 'rmse', 'mae', 'mape', 'mdape', 'smape',
 85        'coverage']. The caller's sequence is not modified.
 86    rolling_window: Proportion of data to use in each rolling window for
 87        computing the metrics. Should be in [0, 1] to average.
 88    monthly: monthly=True will compute horizons as numbers of calendar months
 89        from the cutoff date, starting from 0 for the cutoff month.
 90
 91    Returns
 92    -------
 93    Dataframe with a column for each metric, and column 'horizon'
 94    """
 95    valid_metrics = ['mse', 'rmse', 'mae', 'mape', 'mdape', 'smape', 'coverage']
 96    if metrics is None:
 97        metrics = valid_metrics
 98    # Copy so that removing skipped metrics below does not modify the caller's list
 99    metrics = list(metrics)
100    if ('yhat_lower' not in df or 'yhat_upper' not in df) and ('coverage' in metrics):
101        metrics.remove('coverage')
102    if len(set(metrics)) != len(metrics):
103        raise ValueError('Input metrics must be a list of unique values')
104    if not set(metrics).issubset(set(PERFORMANCE_METRICS)):
105        raise ValueError(
106            'Valid values for metrics are: {}'.format(valid_metrics)
107        )
108    df_m = df.copy()
109    if monthly:
110        df_m['horizon'] = df_m['ds'].dt.to_period('M').astype(int) - df_m['cutoff'].dt.to_period('M').astype(int)
111    else:
112        df_m['horizon'] = df_m['ds'] - df_m['cutoff']
113    df_m.sort_values('horizon', inplace=True)
114    if 'mape' in metrics and df_m['y'].abs().min() < 1e-8:
115        logger.info('Skipping MAPE because y close to 0')
116        metrics.remove('mape')
117    if len(metrics) == 0:
118        return None
119    w = int(rolling_window * df_m.shape[0])
120    if w >= 0:
121        w = max(w, 1)
122        w = min(w, df_m.shape[0])
123    # Compute all metrics
124    dfs = {}
125    for metric in metrics:
126        dfs[metric] = PERFORMANCE_METRICS[metric](df_m, w)
127    res = dfs[metrics[0]]
128    for i in range(1, len(metrics)):
129        res_m = dfs[metrics[i]]
130        assert np.array_equal(res['horizon'].values, res_m['horizon'].values)
131        res[metrics[i]] = res_m[metrics[i]]
132    return res

135def rolling_mean_by_h(x: np.ndarray, h: np.ndarray, w: int, name: str) -> pd.DataFrame:
136    """Compute a rolling mean of x, after first aggregating by h.
137
138    Right-aligned. Computes a single mean for each unique value of h. Each
139    mean is over at least w samples.
140
141    Parameters
142    ----------
143    x: Array.
144    h: Array of horizon for each value in x.
145    w: Integer window size (number of elements).
146    name: Name for metric in result dataframe
147
148    Returns
149    -------
150    Dataframe with columns horizon and name, the rolling mean of x.
151    """
152    # Aggregate over h
153    df = pd.DataFrame({'x': x, 'h': h})
154    df2 = (
155        df.groupby('h').agg(['sum', 'count']).reset_index().sort_values('h')
156    )
157    xs = df2['x']['sum'].values
158    ns = df2['x']['count'].values
159    hs = df2.h.values
160
161    trailing_i = len(df2) - 1
162    x_sum = 0
163    n_sum = 0
164    # We don't know output size but it is bounded by len(df2)
165    res_x = np.empty(len(df2))
166
167    # Start from the right and work backwards
168    for i in range(len(df2) - 1, -1, -1):
169        x_sum += xs[i]
170        n_sum += ns[i]
171        while n_sum >= w:
172            # Include points from the previous horizon. All of them if still
173            # less than w, otherwise weight the mean by the difference
174            excess_n = n_sum - w
175            excess_x = excess_n * xs[i] / ns[i]
176            res_x[trailing_i] = (x_sum - excess_x)/ w
177            x_sum -= xs[trailing_i]
178            n_sum -= ns[trailing_i]
179            trailing_i -= 1
180
181    res_h = hs[(trailing_i + 1):]
182    res_x = res_x[(trailing_i + 1):]
183
184    return pd.DataFrame({'horizon': res_h, name: res_x})

188def rolling_median_by_h(x: np.ndarray, h: np.ndarray, w: int, name: str) -> pd.DataFrame:
189    """Compute a rolling median of x, after first aggregating by h.
190
191    Right-aligned. Computes a single median for each unique value of h. Each
192    median is over at least w samples.
193
194    For each h where there are fewer than w samples, we take samples from the previous h,
195    moving backwards. (In other words, we ~ assume that the x's are shuffled within each h.)
196
197    Parameters
198    ----------
199    x: Array.
200    h: Array of horizon for each value in x.
201    w: Integer window size (number of elements).
202    name: Name for metric in result dataframe
203
204    Returns
205    -------
206    Dataframe with columns horizon and name, the rolling median of x.
207    """
208    # Aggregate over h
209    df = pd.DataFrame({'x': x, 'h': h})
210    grouped = df.groupby('h')
211    df2 = grouped.size().reset_index().sort_values('h')
212    hs = df2['h']
213
214    res_h = []
215    res_x = []
216    # Start from the right and work backwards
217    i = len(hs) - 1
218    while i >= 0:
219        h_i = hs[i]
220        xs = grouped.get_group(h_i).x.tolist()
221
222        # wrap in array so this works if h is pandas Series with custom index or numpy array
223        next_idx_to_add = np.array(h == h_i).argmax() - 1
224        while (len(xs) < w) and (next_idx_to_add >= 0):
225            # Include points from the previous horizon. All of them if still
226            # less than w, otherwise just enough to get to w.
227            xs.append(x[next_idx_to_add])
228            next_idx_to_add -= 1
229        if len(xs) < w:
230            # Ran out of points before getting enough.
231            break
232        res_h.append(hs[i])
233        res_x.append(np.median(xs))
234        i -= 1
235    res_h.reverse()
236    res_x.reverse()
237    return pd.DataFrame({'horizon': res_h, name: res_x})

249    Parameters

265        name='mse',

360@register_performance_metric
361def smape(df: pd.DataFrame, w: int) -> pd.DataFrame:
362    """Symmetric mean absolute percentage error
363    based on Chen and Yang (2004) formula
364
365    Parameters
366    ----------
367    df: Cross-validation results dataframe.
368    w: Aggregation window size.
369
370    Returns
371    -------
372    Dataframe with columns horizon and smape.
373    """
374    sape = (df['y'] - df['yhat']).abs() / ((np.abs(df['y']) + np.abs(df['yhat'])) / 2)
375    sape = sape.fillna(0)
376    if w < 0:
377        return pd.DataFrame({'horizon': df['horizon'], 'smape': sape})
378    return rolling_mean_by_h(
379        x=cast(np.ndarray, sape.values),
380        h=cast(np.ndarray, df['horizon'].values),
381        w=w,
382        name='smape',
383    )

190    """Compute a rolling median of x, after first aggregating by h.
191
192    Right-aligned. Computes a single median for each unique value of h. Each
193    median is over at least w samples.
194
195    For each h where there are fewer than w samples, we take samples from the previous h,
196    moving backwards. (In other words, we ~ assume that the x's are shuffled within each h.)
197
198    Parameters
199    ----------
200    x: Array.
201    h: Array of horizon for each value in x.
202    w: Integer window size (number of elements).
203    name: Name for metric in result dataframe
204
205    Returns
206    -------
207    Dataframe with columns horizon and name, the rolling median of x.
208    """
209    # Aggregate over h
210    df = pd.DataFrame({'x': x, 'h': h})
211    grouped = df.groupby('h')
212    df2 = grouped.size().reset_index().sort_values('h')
213    hs = df2['h']
214
215    res_h = []
216    res_x = []
217    # Start from the right and work backwards
218    i = len(hs) - 1
219    while i >= 0:
220        h_i = hs[i]
221        xs = grouped.get_group(h_i).x.tolist()
222
223        # wrap in array so this works if h is pandas Series with custom index or numpy array
224        next_idx_to_add = np.array(h == h_i).argmax() - 1
225        while (len(xs) < w) and (next_idx_to_add >= 0):
226            # Include points from the previous horizon. All of them if still
227            # less than w, otherwise just enough to get to w.
228            xs.append(x[next_idx_to_add])
229            next_idx_to_add -= 1
230        if len(xs) < w:
231            # Ran out of points before getting enough.
232            break
233        res_h.append(hs[i])
234        res_x.append(np.median(xs))
235        i -= 1
236    res_h.reverse()
237    res_x.reverse()
238    return pd.DataFrame({'horizon': res_h, name: res_x})
