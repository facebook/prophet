 47def generate_cutoffs(
 48    df: pd.DataFrame,
 49    horizon: pd.Timedelta,
 50    initial: pd.Timedelta,
 51    period: pd.Timedelta,
 52) -> list[pd.Timestamp]:
 53    """Generate cutoff dates
 54
 55    Parameters
 56    ----------
 57    df: pd.DataFrame with historical data.
 58    horizon: pd.Timedelta forecast horizon.
 59    initial: pd.Timedelta window of the initial forecast period.
 60    period: pd.Timedelta simulated forecasts are done with this period.
 61
 62    Returns
 63    -------
 64    list of pd.Timestamp
 65    """
 66    # Last cutoff is 'latest date in data - horizon' date
 67    cutoff = df['ds'].max() - horizon
 68    if cutoff < df['ds'].min():
 69        raise ValueError('Less data than horizon.')
 70    result = [cutoff]
 71    while result[-1] >= min(df['ds']) + initial:
 72        cutoff -= period
 73        # If data does not exist in data range (cutoff, cutoff + horizon]
 74        if not (((df['ds'] > cutoff) & (df['ds'] <= cutoff + horizon)).any()):
 75            # Next cutoff point is 'last date before cutoff in data - horizon'
 76            if cutoff > df['ds'].min():
 77                closest_date = df[df['ds'] <= cutoff].max()['ds']
 78                cutoff = closest_date - horizon
 79            # else no data left, leave cutoff as is, it will be dropped.
 80        result.append(cutoff)
 81    result = result[:-1]
 82    if len(result) == 0:
 83        raise ValueError(
 84            'Less data than horizon after initial window. '
 85            'Make horizon or initial shorter.'
 86        )
 87    logger.info('Making {} forecasts with cutoffs between {} and {}'.format(
 88        len(result), result[-1], result[0]
 89    ))
 90    return list(reversed(result))

 93def cross_validation(
 94    model: Prophet,
 95    horizon: str | pd.Timedelta,
 96    period: str | pd.Timedelta | None = None,
 97    initial: str | pd.Timedelta | None = None,
 98    parallel: Literal["processes", "threads", "dask"] | _SupportsMap | None = None,
 99    cutoffs: list[pd.Timestamp] | None = None,
100    disable_tqdm: bool = False,
101    extra_output_columns: str | list[str] | None = None,
102) -> pd.DataFrame:
103    """Cross-Validation for time series.
104
105    Computes forecasts from historical cutoff points, which user can input.
106    If not provided, begins from (end - horizon) and works backwards, making
107    cutoffs with a spacing of period until initial is reached.
108
109    When period is equal to the time interval of the data, this is the
110    technique described in https://robjhyndman.com/hyndsight/tscv/ .
111
112    Parameters
113    ----------
114    model: Prophet class object. Fitted Prophet model.
115    horizon: string with pd.Timedelta compatible style, e.g., '5 days',
116        '3 hours', '10 seconds'.
117    period: string with pd.Timedelta compatible style. Simulated forecast will
118        be done at every this period. If not provided, 0.5 * horizon is used.
119    initial: string with pd.Timedelta compatible style. The first training
120        period will include at least this much data. If not provided,
121        3 * horizon is used.
122    cutoffs: list of pd.Timestamp specifying cutoffs to be used during
123        cross validation. If not provided, they are generated as described
124        above.
125    parallel: {None, 'processes', 'threads', 'dask', object}
126        How to parallelize the forecast computation. By default no parallelism
127        is used.
128
129        * None: No parallelism.
130        * 'processes': Parallelize with concurrent.futures.ProcessPoolExecutor.
131        * 'threads': Parallelize with concurrent.futures.ThreadPoolExecutor.
132            Note that some operations currently hold Python's Global Interpreter
133            Lock, so parallelizing with threads may be slower than training
134            sequentially.
135        * 'dask': Parallelize with Dask.
136           This requires that a dask.distributed Client be created.
137        * object: Any instance with a `.map` method. This method will
138          be called with :func:`single_cutoff_forecast` and a sequence of
139          iterables where each element is the tuple of arguments to pass to
140          :func:`single_cutoff_forecast`
141
142          .. code-block::
143
144             class MyBackend:
145                 def map(self, func, *iterables):
146                     results = [
147                        func(*args)
148                        for args in zip(*iterables)
149                     ]
150                     return results
151
152    disable_tqdm: if True it disables the progress bar that would otherwise show up when parallel=None
153    extra_output_columns: A String or List of Strings e.g. 'trend' or ['trend'].
154         Additional columns to 'yhat' and 'ds' to be returned in output.
155
156    Returns
157    -------
158    A pd.DataFrame with the forecast, actual value and cutoff.
159    """
160
161    if model.history is None:
162        raise Exception('Model has not been fit. Fitting the model provides contextual parameters for cross validation.')
163
164    df = model.history.copy().reset_index(drop=True)
165    horizon = pd.Timedelta(horizon)
166    predict_columns = ['ds', 'yhat']
167
168    if model.uncertainty_samples:
169        predict_columns.extend(['yhat_lower', 'yhat_upper'])
170
171    if extra_output_columns is not None:
172        if isinstance(extra_output_columns, str):
173            extra_output_columns = [extra_output_columns]
174        predict_columns.extend([c for c in extra_output_columns if c not in predict_columns])
175
176    # Identify the largest seasonality period
177    period_max = 0.
178    for s in model.seasonalities.values():
179        period_max = max(period_max, s['period'])
180    seasonality_dt = pd.Timedelta(str(period_max) + ' days')
181
182    if cutoffs is None:
183        # Set period
184        period = 0.5 * horizon if period is None else pd.Timedelta(period)
185
186        # Set initial
187        initial = (
188            max(3 * horizon, seasonality_dt) if initial is None
189            else pd.Timedelta(initial)
190        )
191
192        # Compute Cutoffs
193        cutoffs = generate_cutoffs(df, horizon, initial, period)
194    else:
195        # add validation of the cutoff to make sure that the min cutoff is strictly greater than the min date in the history
196        if min(cutoffs) <= df['ds'].min():
197            raise ValueError("Minimum cutoff value is not strictly greater than min date in history")
198        # max value of cutoffs is <= (end date minus horizon)
199        end_date_minus_horizon = df['ds'].max() - horizon
200        if max(cutoffs) > end_date_minus_horizon:
201            raise ValueError("Maximum cutoff value is greater than end date minus horizon, no value for cross-validation remaining")
202        initial = cutoffs[0] - df['ds'].min()
203
204    # Check if the initial window
205    # (that is, the amount of time between the start of the history and the first cutoff)
206    # is less than the maximum seasonality period
207    if initial < seasonality_dt:
208            msg = 'Seasonality has period of {} days '.format(period_max)
209            msg += 'which is larger than initial window. '
210            msg += 'Consider increasing initial.'
211            logger.warning(msg)
212
213    if parallel:
214        valid = {"threads", "processes", "dask"}
215
216        if parallel == "threads":
217            pool = concurrent.futures.ThreadPoolExecutor()
218        elif parallel == "processes":
219            if sys.platform.startswith("win") or sys.platform == "darwin":
220                ctx = multiprocessing.get_context("spawn")
221            else:
222                ctx = multiprocessing.get_context("forkserver")
223            pool = concurrent.futures.ProcessPoolExecutor(mp_context=ctx)
224        elif parallel == "dask":
225            try:
226                from dask.distributed import get_client
227            except ImportError as e:
228                raise ImportError("parallel='dask' requires the optional "
229                                  "dependency dask.") from e
230            pool = get_client()
231            # delay df and model to avoid large objects in task graph.
232            df, model = pool.scatter([df, model])
233        elif hasattr(parallel, "map"):
234            pool = parallel
235        else:
236            msg = ("'parallel' should be one of {} for an instance with a "
237                   "'map' method".format(', '.join(valid)))
238            raise ValueError(msg)
239
240        iterables = ((df, model, cutoff, horizon, predict_columns)
241                     for cutoff in cutoffs)
242        iterables = zip(*iterables)
243
244        logger.info("Applying in parallel with %s", pool)
245        predicts = pool.map(single_cutoff_forecast, *iterables)
246        if parallel == "dask":
247            # convert Futures to DataFrames
248            predicts = cast("dd.Client", pool).gather(predicts)
249
250    else:
251        predicts = [
252            single_cutoff_forecast(df, model, cutoff, horizon, predict_columns)
253            for cutoff in (tqdm(cutoffs) if not disable_tqdm else cutoffs)
254        ]
255
256    # Combine all predicted pd.DataFrame into one pd.DataFrame
257    return pd.concat(cast("Iterable[Any]", predicts), axis=0).reset_index(drop=True)

260def single_cutoff_forecast(
261    df: pd.DataFrame,
262    model: Prophet,
263    cutoff: pd.Timestamp,
264    horizon: pd.Timedelta,
265    predict_columns: list[str],
266) -> pd.DataFrame:
267    """Forecast for single cutoff. Used in cross validation function
268    when evaluating for multiple cutoffs either sequentially or in parallel.
269
270    Parameters
271    ----------
272    df: pd.DataFrame.
273        DataFrame with history to be used for single
274        cutoff forecast.
275    model: Prophet model object.
276    cutoff: pd.Timestamp cutoff date.
277        Simulated Forecast will start from this date.
278    horizon: pd.Timedelta forecast horizon.
279    predict_columns: List of strings e.g. ['ds', 'yhat'].
280        Columns with date and forecast to be returned in output.
281
282    Returns
283    -------
284    A pd.DataFrame with the forecast, actual value and cutoff.
285
286    """
287
288    # Generate new object with copying fitting options
289    m = prophet_copy(model, cutoff)
290    # Train model
291    history_c = df[df['ds'] <= cutoff]
292    if history_c.shape[0] < 2:
293        raise Exception(
294            'Less than two datapoints before cutoff. '
295            'Increase initial window.'
296        )
297    m.fit(history_c, **model.fit_kwargs)
298    # Calculate yhat
299    index_predicted = (df['ds'] > cutoff) & (df['ds'] <= cutoff + horizon)
300    # Get the columns for the future dataframe
301    columns = ['ds']
302    if m.growth == 'logistic':
303        columns.append('cap')
304        if m.logistic_floor:
305            columns.append('floor')
306    columns.extend(m.extra_regressors.keys())
307    columns.extend([
308        props['condition_name']
309        for props in m.seasonalities.values()
310        if props['condition_name'] is not None])
311    yhat = m.predict(df[index_predicted][columns])
312    # Merge yhat(predicts), y(df, original data) and cutoff
313
314    assert m.stan_backend
315    m.stan_backend.cleanup()
316
317    return pd.concat([
318        yhat[predict_columns],
319        df[index_predicted][['y']].reset_index(drop=True),
320        pd.DataFrame({'cutoff': [cutoff] * len(yhat)})
321    ], axis=1)

324def prophet_copy(m: _ModelT, cutoff: pd.Timestamp | None = None) -> _ModelT:
325    """Copy Prophet object
326
327    Parameters
328    ----------
329    m: Prophet model.
330    cutoff: pd.Timestamp or None, default None.
331        cutoff Timestamp for changepoints member variable.
332        changepoints are only retained if 'changepoints <= cutoff'
333
334    Returns
335    -------
336    Prophet class object with the same parameter with model variable
337    """
338    if m.history is None:
339        raise Exception('This is for copying a fitted Prophet object.')
340
341    if m.specified_changepoints:
342        changepoints = m.changepoints
343        if cutoff is not None:
344            # Filter change points '< cutoff'
345            last_history_date = max(m.history['ds'][m.history['ds'] <= cutoff])
346            assert changepoints is not None
347            changepoints = changepoints[changepoints < last_history_date]
348    else:
349        changepoints = None
350
351    # Auto seasonalities are set to False because they are already set in
352    # m.seasonalities.
353    m2 = m.__class__(
354        growth=m.growth,
355        n_changepoints=m.n_changepoints,
356        changepoint_range=m.changepoint_range,
357        changepoints=changepoints,
358        yearly_seasonality=False,
359        weekly_seasonality=False,
360        daily_seasonality=False,
361        holidays=m.holidays,
362        holidays_mode=m.holidays_mode,
363        seasonality_mode=m.seasonality_mode,
364        seasonality_prior_scale=m.seasonality_prior_scale,
365        changepoint_prior_scale=m.changepoint_prior_scale,
366        holidays_prior_scale=m.holidays_prior_scale,
367        mcmc_samples=m.mcmc_samples,
368        interval_width=m.interval_width,
369        uncertainty_samples=m.uncertainty_samples,
370        stan_backend=(
371            m.stan_backend.get_type() if m.stan_backend is not None
372            else None
373        ),
374    )
375    m2.extra_regressors = deepcopy(m.extra_regressors)
376    m2.seasonalities = deepcopy(m.seasonalities)
377    m2.country_holidays = deepcopy(m.country_holidays)
378    return m2

382def register_performance_metric(func: _PerformanceMetricT) -> _PerformanceMetricT:
383    """Register custom performance metric
384
385    Parameters that your metric should contain
386    ----------
387    df: Cross-validation results dataframe.
388    w: Aggregation window size.
389
390    Registered metric should return following
391    -------
392    Dataframe with columns horizon and metric.
393    """
394    PERFORMANCE_METRICS[func.__name__] = func
395    return func

398def performance_metrics(
399    df: pd.DataFrame,
400    metrics: list[str] | None = None,
401    rolling_window: float = 0.1,
402    monthly: bool = False,
403) -> pd.DataFrame | None:
404    """Compute performance metrics from cross-validation results.
405
406    Computes a suite of performance metrics on the output of cross-validation.
407    By default, the following metrics are included:
408    'mse': mean squared error
409    'rmse': root mean squared error
410    'mae': mean absolute error
411    'mape': mean absolute percent error
412    'mdape': median absolute percent error
413    'smape': symmetric mean absolute percentage error
414    'coverage': coverage of the upper and lower intervals
415
416    A subset of these can be specified by passing a list of names as the
417    `metrics` argument.
418
419    Metrics are calculated over a rolling window of cross validation
420    predictions, after sorting by horizon. Averaging is first done within each
421    value of horizon, and then across horizons as needed to reach the window
422    size. The size of that window (number of simulated forecast points) is
423    determined by the rolling_window argument, which specifies a proportion of
424    simulated forecast points to include in each window. rolling_window=0 will
425    compute it separately for each horizon. The default of rolling_window=0.1
426    will use 10% of the rows in df in each window. rolling_window=1 will
427    compute the metric across all simulated forecast points. The results are
428    set to the right edge of the window.
429
430    If rolling_window < 0, then metrics are computed at each datapoint with no
431    averaging (i.e., 'mse' will actually be squared error with no mean).
432
433    The output is a dataframe containing column 'horizon' along with columns
434    for each of the metrics computed.
435
436    Parameters
437    ----------
438    df: The dataframe returned by cross_validation.
439    metrics: A list of performance metrics to compute. If not provided, will
440        use ['mse', 'rmse', 'mae', 'mape', 'mdape', 'smape', 'coverage'].
441    rolling_window: Proportion of data to use in each rolling window for
442        computing the metrics. Should be in [0, 1] to average.
443    monthly: monthly=True will compute horizons as numbers of calendar months
444        from the cutoff date, starting from 0 for the cutoff month.
445
446    Returns
447    -------
448    Dataframe with a column for each metric, and column 'horizon'
449    """
450    valid_metrics = ['mse', 'rmse', 'mae', 'mape', 'mdape', 'smape', 'coverage']
451    if metrics is None:
452        metrics = valid_metrics
453    if ('yhat_lower' not in df or 'yhat_upper' not in df) and ('coverage' in metrics):
454        metrics.remove('coverage')
455    if len(set(metrics)) != len(metrics):
456        raise ValueError('Input metrics must be a list of unique values')
457    if not set(metrics).issubset(set(PERFORMANCE_METRICS)):
458        raise ValueError(
459            'Valid values for metrics are: {}'.format(valid_metrics)
460        )
461    df_m = df.copy()
462    if monthly:
463        df_m['horizon'] = df_m['ds'].dt.to_period('M').astype(int) - df_m['cutoff'].dt.to_period('M').astype(int)
464    else:
465        df_m['horizon'] = df_m['ds'] - df_m['cutoff']
466    df_m.sort_values('horizon', inplace=True)
467    if 'mape' in metrics and df_m['y'].abs().min() < 1e-8:
468        logger.info('Skipping MAPE because y close to 0')
469        metrics.remove('mape')
470    if len(metrics) == 0:
471        return None
472    w = int(rolling_window * df_m.shape[0])
473    if w >= 0:
474        w = max(w, 1)
475        w = min(w, df_m.shape[0])
476    # Compute all metrics
477    dfs = {}
478    for metric in metrics:
479        dfs[metric] = PERFORMANCE_METRICS[metric](df_m, w)
480    res = dfs[metrics[0]]
481    for i in range(1, len(metrics)):
482        res_m = dfs[metrics[i]]
483        assert np.array_equal(res['horizon'].values, res_m['horizon'].values)
484        res[metrics[i]] = res_m[metrics[i]]
485    return res

488def rolling_mean_by_h(x: np.ndarray, h: np.ndarray, w: int, name: str) -> pd.DataFrame:
489    """Compute a rolling mean of x, after first aggregating by h.
490
491    Right-aligned. Computes a single mean for each unique value of h. Each
492    mean is over at least w samples.
493
494    Parameters
495    ----------
496    x: Array.
497    h: Array of horizon for each value in x.
498    w: Integer window size (number of elements).
499    name: Name for metric in result dataframe
500
501    Returns
502    -------
503    Dataframe with columns horizon and name, the rolling mean of x.
504    """
505    # Aggregate over h
506    df = pd.DataFrame({'x': x, 'h': h})
507    df2 = (
508        df.groupby('h').agg(['sum', 'count']).reset_index().sort_values('h')
509    )
510    xs = df2['x']['sum'].values
511    ns = df2['x']['count'].values
512    hs = df2.h.values
513
514    trailing_i = len(df2) - 1
515    x_sum = 0
516    n_sum = 0
517    # We don't know output size but it is bounded by len(df2)
518    res_x = np.empty(len(df2))
519
520    # Start from the right and work backwards
521    for i in range(len(df2) - 1, -1, -1):
522        x_sum += xs[i]
523        n_sum += ns[i]
524        while n_sum >= w:
525            # Include points from the previous horizon. All of them if still
526            # less than w, otherwise weight the mean by the difference
527            excess_n = n_sum - w
528            excess_x = excess_n * xs[i] / ns[i]
529            res_x[trailing_i] = (x_sum - excess_x)/ w
530            x_sum -= xs[trailing_i]
531            n_sum -= ns[trailing_i]
532            trailing_i -= 1
533
534    res_h = hs[(trailing_i + 1):]
535    res_x = res_x[(trailing_i + 1):]
536
537    return pd.DataFrame({'horizon': res_h, name: res_x})

541def rolling_median_by_h(x: np.ndarray, h: np.ndarray, w: int, name: str) -> pd.DataFrame:
542    """Compute a rolling median of x, after first aggregating by h.
543
544    Right-aligned. Computes a single median for each unique value of h. Each
545    median is over at least w samples.
546
547    For each h where there are fewer than w samples, we take samples from the previous h,
548    moving backwards. (In other words, we ~ assume that the x's are shuffled within each h.)
549
550    Parameters
551    ----------
552    x: Array.
553    h: Array of horizon for each value in x.
554    w: Integer window size (number of elements).
555    name: Name for metric in result dataframe
556
557    Returns
558    -------
559    Dataframe with columns horizon and name, the rolling median of x.
560    """
561    # Aggregate over h
562    df = pd.DataFrame({'x': x, 'h': h})
563    grouped = df.groupby('h')
564    df2 = grouped.size().reset_index().sort_values('h')
565    hs = df2['h']
566
567    res_h = []
568    res_x = []
569    # Start from the right and work backwards
570    i = len(hs) - 1
571    while i >= 0:
572        h_i = hs[i]
573        xs = grouped.get_group(h_i).x.tolist()
574
575        # wrap in array so this works if h is pandas Series with custom index or numpy array
576        next_idx_to_add = np.array(h == h_i).argmax() - 1
577        while (len(xs) < w) and (next_idx_to_add >= 0):
578            # Include points from the previous horizon. All of them if still
579            # less than w, otherwise just enough to get to w.
580            xs.append(x[next_idx_to_add])
581            next_idx_to_add -= 1
582        if len(xs) < w:
583            # Ran out of points before getting enough.
584            break
585        res_h.append(hs[i])
586        res_x.append(np.median(xs))
587        i -= 1
588    res_h.reverse()
589    res_x.reverse()
590    return pd.DataFrame({'horizon': res_h, name: res_x})

593# The functions below specify performance metrics for cross-validation results.
594# Each takes as input the output of cross_validation, and returns the statistic
595# as a dataframe, given a window size for rolling aggregation.

598@register_performance_metric
599def mse(df: pd.DataFrame, w: int) -> pd.DataFrame:
600    """Mean squared error
601
602    Parameters
603    ----------
604    df: Cross-validation results dataframe.
605    w: Aggregation window size.
606
607    Returns
608    -------
609    Dataframe with columns horizon and mse.
610    """
611    se = (df['y'] - df['yhat']) ** 2
612    if w < 0:
613        return pd.DataFrame({'horizon': df['horizon'], 'mse': se})
614    return rolling_mean_by_h(
615        x=cast(np.ndarray, se.values),
616        h=cast(np.ndarray, df['horizon'].values),
617        w=w,
618        name='mse',
619    )

622@register_performance_metric
623def rmse(df: pd.DataFrame, w: int) -> pd.DataFrame:
624    """Root mean squared error
625
626    Parameters
627    ----------
628    df: Cross-validation results dataframe.
629    w: Aggregation window size.
630
631    Returns
632    -------
633    Dataframe with columns horizon and rmse.
634    """
635    res = mse(df, w)
636    res['mse'] = np.sqrt(res['mse'])
637    res.rename({'mse': 'rmse'}, axis='columns', inplace=True)
638    return res

641@register_performance_metric
642def mae(df: pd.DataFrame, w: int) -> pd.DataFrame:
643    """Mean absolute error
644
645    Parameters
646    ----------
647    df: Cross-validation results dataframe.
648    w: Aggregation window size.
649
650    Returns
651    -------

def generate_cutoffs( df: pandas.DataFrame, horizon: pandas.Timedelta, initial: pandas.Timedelta, period: pandas.Timedelta) -> list[pandas.Timestamp]: 
48def generate_cutoffs(
49    df: pd.DataFrame,
50    horizon: pd.Timedelta,
51    initial: pd.Timedelta,
52    period: pd.Timedelta,
53) -> list[pd.Timestamp]:
54    """Generate cutoff dates
55
56    Parameters
57    ----------
58    df: pd.DataFrame with historical data.
59    horizon: pd.Timedelta forecast horizon.
60    initial: pd.Timedelta window of the initial forecast period.
61    period: pd.Timedelta simulated forecasts are done with this period.
62
63    Returns
64    -------
65    list of pd.Timestamp
66    """
67    # Last cutoff is 'latest date in data - horizon' date
68    cutoff = df['ds'].max() - horizon
69    if cutoff < df['ds'].min():
70        raise ValueError('Less data than horizon.')
71    result = [cutoff]
72    while result[-1] >= min(df['ds']) + initial:
73        cutoff -= period
74        # If data does not exist in data range (cutoff, cutoff + horizon]
75        if not (((df['ds'] > cutoff) & (df['ds'] <= cutoff + horizon)).any()):
76            # Next cutoff point is 'last date before cutoff in data - horizon'
77            if cutoff > df['ds'].min():
78                closest_date = df[df['ds'] <= cutoff].max()['ds']
79                cutoff = closest_date - horizon
80            # else no data left, leave cutoff as is, it will be dropped.
81        result.append(cutoff)
82    result = result[:-1]
83    if len(result) == 0:
84        raise ValueError(
85            'Less data than horizon after initial window. '
86            'Make horizon or initial shorter.'
87        )
88    logger.info('Making {} forecasts with cutoffs between {} and {}'.format(
89        len(result), result[-1], result[0]
90    ))
91    return list(reversed(result))

406
407    Computes a suite of performance metrics on the output of cross-validation.
408    By default, the following metrics are included:
409    'mse': mean squared error
410    'rmse': root mean squared error
411    'mae': mean absolute error
412    'mape': mean absolute percent error
413    'mdape': median absolute percent error
414    'smape': symmetric mean absolute percentage error
415    'coverage': coverage of the upper and lower intervals
416
417    A subset of these can be specified by passing a list of names as the
418    `metrics` argument.
419
420    Metrics are calculated over a rolling window of cross validation
421    predictions, after sorting by horizon. Averaging is first done within each
422    value of horizon, and then across horizons as needed to reach the window
423    size. The size of that window (number of simulated forecast points) is
424    determined by the rolling_window argument, which specifies a proportion of
425    simulated forecast points to include in each window. rolling_window=0 will
426    compute it separately for each horizon. The default of rolling_window=0.1
427    will use 10% of the rows in df in each window. rolling_window=1 will
428    compute the metric across all simulated forecast points. The results are
429    set to the right edge of the window.
430

432    averaging (i.e., 'mse' will actually be squared error with no mean).
433

435    for each of the metrics computed.
436
437    Parameters
438    ----------
439    df: The dataframe returned by cross_validation.

441        use ['mse', 'rmse', 'mae', 'mape', 'mdape', 'smape', 'coverage'].
442    rolling_window: Proportion of data to use in each rolling window for
443        computing the metrics. Should be in [0, 1] to average.

455        metrics.remove('coverage')
456    if len(set(metrics)) != len(metrics):
457        raise ValueError('Input metrics must be a list of unique values')
458    if not set(metrics).issubset(set(PERFORMANCE_METRICS)):
459        raise ValueError(

470        metrics.remove('mape')

482    for i in range(1, len(metrics)):
483        res_m = dfs[metrics[i]]
484        assert np.array_equal(res['horizon'].values, res_m['horizon'].values)
485        res[metrics[i]] = res_m[metrics[i]]

Compute performance metrics from cross-validation results.

Computes a suite of performance metrics on the output of cross-validation. By default, the following metrics are included: 'mse': mean squared error 'rmse': root mean squared error 'mae': mean absolute error 'mape': mean absolute percent error 'mdape': median absolute percent error 'smape': symmetric mean absolute percentage error 'coverage': coverage of the upper and lower intervals

A subset of these can be specified by passing a list of names as the metrics argument.

Metrics are calculated over a rolling window of cross validation predictions, after sorting by horizon. Averaging is first done within each value of horizon, and then across horizons as needed to reach the window size. The size of that window (number of simulated forecast points) is determined by the rolling_window argument, which specifies a proportion of simulated forecast points to include in each window. rolling_window=0 will compute it separately for each horizon. The default of rolling_window=0.1 will use 10% of the rows in df in each window. rolling_window=1 will compute the metric across all simulated forecast points. The results are set to the right edge of the window.

If rolling_window < 0, then metrics are computed at each datapoint with no averaging (i.e., 'mse' will actually be squared error with no mean).

The output is a dataframe containing column 'horizon' along with columns for each of the metrics computed.

Parameters

  • df (The dataframe returned by cross_validation.):

  • metrics (A list of performance metrics to compute. If not provided, will): use ['mse', 'rmse', 'mae', 'mape', 'mdape', 'smape', 'coverage'].

  • rolling_window (Proportion of data to use in each rolling window for): computing the metrics. Should be in [0, 1] to average.
  • monthly (monthly=True will compute horizons as numbers of calendar months): from the cutoff date, starting from 0 for the cutoff month.
Returns
  • Dataframe with a column for each metric, and column 'horizon'
def rolling_mean_by_h( x: numpy.ndarray, h: numpy.ndarray, w: int, name: str) -> pandas.DataFrame: 
489def rolling_mean_by_h(x: np.ndarray, h: np.ndarray, w: int, name: str) -> pd.DataFrame:
490    """Compute a rolling mean of x, after first aggregating by h.
491
492    Right-aligned. Computes a single mean for each unique value of h. Each
493    mean is over at least w samples.
494
495    Parameters
496    ----------
497    x: Array.
498    h: Array of horizon for each value in x.
499    w: Integer window size (number of elements).
500    name: Name for metric in result dataframe
501
502    Returns
503    -------
504    Dataframe with columns horizon and name, the rolling mean of x.
505    """
506    # Aggregate over h
507    df = pd.DataFrame({'x': x, 'h': h})
508    df2 = (
509        df.groupby('h').agg(['sum', 'count']).reset_index().sort_values('h')
510    )
511    xs = df2['x']['sum'].values
512    ns = df2['x']['count'].values
513    hs = df2.h.values
514
515    trailing_i = len(df2) - 1
516    x_sum = 0
517    n_sum = 0
518    # We don't know output size but it is bounded by len(df2)
519    res_x = np.empty(len(df2))
520
521    # Start from the right and work backwards
522    for i in range(len(df2) - 1, -1, -1):
523        x_sum += xs[i]
524        n_sum += ns[i]
525        while n_sum >= w:
526            # Include points from the previous horizon. All of them if still
527            # less than w, otherwise weight the mean by the difference
528            excess_n = n_sum - w
529            excess_x = excess_n * xs[i] / ns[i]
530            res_x[trailing_i] = (x_sum - excess_x)/ w
531            x_sum -= xs[trailing_i]
532            n_sum -= ns[trailing_i]
533            trailing_i -= 1
534
535    res_h = hs[(trailing_i + 1):]
536    res_x = res_x[(trailing_i + 1):]
537
538    return pd.DataFrame({'horizon': res_h, name: res_x})

def rolling_median_by_h( x: numpy.ndarray, h: numpy.ndarray, w: int, name: str) -> pandas.DataFrame: 
542def rolling_median_by_h(x: np.ndarray, h: np.ndarray, w: int, name: str) -> pd.DataFrame:
543    """Compute a rolling median of x, after first aggregating by h.
544
545    Right-aligned. Computes a single median for each unique value of h. Each
546    median is over at least w samples.
547
548    For each h where there are fewer than w samples, we take samples from the previous h,
549    moving backwards. (In other words, we ~ assume that the x's are shuffled within each h.)
550
551    Parameters
552    ----------
553    x: Array.
554    h: Array of horizon for each value in x.
555    w: Integer window size (number of elements).
556    name: Name for metric in result dataframe
557
558    Returns
559    -------
560    Dataframe with columns horizon and name, the rolling median of x.
561    """
562    # Aggregate over h
563    df = pd.DataFrame({'x': x, 'h': h})
564    grouped = df.groupby('h')
565    df2 = grouped.size().reset_index().sort_values('h')
566    hs = df2['h']
567
568    res_h = []
569    res_x = []
570    # Start from the right and work backwards
571    i = len(hs) - 1
572    while i >= 0:
573        h_i = hs[i]
574        xs = grouped.get_group(h_i).x.tolist()
575
576        # wrap in array so this works if h is pandas Series with custom index or numpy array
577        next_idx_to_add = np.array(h == h_i).argmax() - 1
578        while (len(xs) < w) and (next_idx_to_add >= 0):
579            # Include points from the previous horizon. All of them if still
580            # less than w, otherwise just enough to get to w.
581            xs.append(x[next_idx_to_add])
582            next_idx_to_add -= 1
583        if len(xs) < w:
584            # Ran out of points before getting enough.
585            break
586        res_h.append(hs[i])
587        res_x.append(np.median(xs))
588        i -= 1
589    res_h.reverse()
590    res_x.reverse()
591    return pd.DataFrame({'horizon': res_h, name: res_x})
